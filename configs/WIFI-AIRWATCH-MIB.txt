WIFI-AIRWATCH-MIB DEFINITIONS ::= BEGIN

IMPORTS
    OBJECT-TYPE, MODULE-IDENTITY, enterprisenotify, enterprise
        FROM SNMPv2-SMI;

wifiAirWatch MODULE-IDENTITY
    LAST-UPDATED "202611170000Z"
    ORGANIZATION "wifi-airwatch Open Source Project"
    CONTACT-INFO "[suspicious link removed]"
    DESCRIPTION  "Management Information Base for Raspberry Pi Wi-Fi Telemetry Probe"
    ::= { enterprise 99999 }

-- Objects / Telemetry OIDs

airwatchObjects OBJECT IDENTIFIER ::= { wifiAirWatch 1 }

airwatchBeaconCount OBJECT-TYPE
    SYNTAX      Counter64
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION "Total number of 802.11 beacon frames captured."
    ::= { airwatchObjects 1 }

airwatchBeaconPercentage OBJECT-TYPE
    SYNTAX      Integer32
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION "Percentage of beacon frames compared to total RF traffic."
    ::= { airwatchObjects 2 }

airwatchChannelCongestion OBJECT-TYPE
    SYNTAX      Integer32
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION "Calculated channel occupancy in percentage."
    ::= { airwatchObjects 3 }

END
